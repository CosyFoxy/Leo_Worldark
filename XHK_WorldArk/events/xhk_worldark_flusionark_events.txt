namespace = xhk_worldark_flusionark

# FLUSION INTEGRATION
# STEP 1) the kaiser is defeated and flusion is still colonizable (checked on monthly pulse)
# STEP 2) scan flusion with the world ark (special project), which starts the situation

# situation stages, entering each stage gives the next reward:
	# stage 1: feasibility analysis
	# stage 2: deck expansions become advanced expansions (+250 housing)
	# stage 3: advanced expansions get -5% upkeep for resource-producing jobs
	# stage 4: advanced expansions get +1% max districts and +2.5% automated workforce
	# complete: capital gets the flusion bulwark deposit (+5% and +2 damage reduction)

# on action event to check to start it
event = {
	id = xhk_worldark_flusionark.001
	is_triggered_only = yes
	hide_window = yes
	fire_only_once = yes

	trigger = {
		has_global_flag = katzenartig_imperium_defeated
		xhk_worldark_flusion_is_integratable = yes
	}

	immediate = {
		every_country = {
			limit = {
				xhk_has_worldark_origin = yes
			}
			country_event = {
				id = xhk_worldark_flusionark.002 days = 5
			}
		}
	}
}

# starts the chain, gives the project
country_event = {
	id = xhk_worldark_flusionark.002
	is_triggered_only = yes
	title = xhk_worldark_flusionark.002.title
	desc = xhk_worldark_flusionark.002.desc
	picture = GFX_evt_city_planet

	trigger = {
		xhk_worldark_flusion_is_integratable = yes
	}

	immediate = {
		begin_event_chain = {
			event_chain = xhk_worldark_flusionark_chain
			target = ROOT
		}
		enable_special_project = {
			name = XHK_WORLDARK_FLUSIONARK_ANALYSIS
			location = event_target:gigaflusion
			owner = root
		}
	}

	option = {
		name = xhk_worldark_flusionark.002.a
	}
}

## special project calls this event to start situation
ship_event = {
	id = xhk_worldark_flusionark.010
	is_triggered_only = yes
	title = xhk_worldark_flusionark.010.title
	desc = xhk_worldark_flusionark.010.desc
	picture = GFX_evt_city_planet

	immediate = {
		owner = {
			end_event_chain = xhk_worldark_flusionark_chain
			start_situation = {
				type = xhk_worldark_situation_flusionark
				target = capital_scope
			}
		}
	}
	option = {
		name = xhk_worldark_flusionark.010.a
		custom_tooltip = xhk_worldark_flusionark.010.att
	}
}
